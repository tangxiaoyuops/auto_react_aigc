## 1. 项目概述

### 1.1 项目定位

DataAgent 是一个企业级AI智能体平台，提供从Agent构建、配置、调试到评测的全生命周期管理能力。平台支持多领域空间隔离，集成知识库、Skill、本体等多种能力，实现企业数据的智能化问答和自动化处理。

### 1.2 核心能力

```
┌─────────────────────────────────────────────────────────────────┐
│                        DataAgent 核心能力                        │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  Agent构建          对话调试          资源管理          效果评测  │
│  ┌──────────┐      ┌──────────┐      ┌──────────┐      ┌──────┐│
│  │ 基础配置  │      │ 实时对话  │      │ 知识库    │      │评测集││
│  │ 能力绑定  │      │ 过程追溯  │      │ Skill库   │      │评测  ││
│  │ 子Agent  │      │ 执行明细  │      │ 提示词    │      │任务  ││
│  │ 模型选择  │      │ 节点详情  │      │ 数据源    │      │报告  ││
│  └──────────┘      └──────────┘      │ 本体管理  │      └──────┘│
│                                      └──────────┘              │
│                                                                  │
└─────────────────────────────────────────────────────────────────┘
```

### 1.3 系统特性

| 特性 | 描述 |
|------|------|
| 多领域空间 | 支持多租户，领域隔离，独立管理 |
| Agent全生命周期 | 创建→配置→调试→上架→评测→优化 |
| 多Agent协作 | 支持子Agent编排，复杂任务分解 |
| 过程可视化 | 对话过程节点追溯，执行明细展示 |
| 资源统一管理 | 知识库、Skill、提示词、数据源集中管理 |
| 能力可扩展 | 支持动态添加知识库、本体、Skill等能力 |

// 事件类型定义
interface AgentEvent {
    type: string;
    timestamp: number;
    data: any;
}

// 事件类型枚举
enum EventType {
    // 运行状态
    RUN_START = "RUN_START",
    RUN_END = "RUN_END",
    RUN_ERROR = "RUN_ERROR",
    
    // 节点事件
    NODE_START = "NODE_START",
    NODE_END = "NODE_END",
    NODE_ERROR = "NODE_ERROR",
    
    // 工具调用
    TOOL_CALL_START = "TOOL_CALL_START",
    TOOL_CALL_ARGS = "TOOL_CALL_ARGS",
    TOOL_RESULT = "TOOL_RESULT",
    
    // 知识检索
    KNOWLEDGE_SEARCH_START = "KNOWLEDGE_SEARCH_START",
    KNOWLEDGE_SEARCH_RESULT = "KNOWLEDGE_SEARCH_RESULT",
    
    // 子Agent
    SUB_AGENT_START = "SUB_AGENT_START",
    SUB_AGENT_END = "SUB_AGENT_END",
    
    // 文本消息
    TEXT_MESSAGE_START = "TEXT_MESSAGE_START",
    TEXT_MESSAGE_CONTENT = "TEXT_MESSAGE_CONTENT",
    TEXT_MESSAGE_END = "TEXT_MESSAGE_END",
    
    // 思考过程
    THOUGHT = "THOUGHT",
    OBSERVATION = "OBSERVATION",
}

// 节点事件数据结构
interface NodeStartEvent {
    node_id: string;
    node_type: "intent_recognition" | "knowledge_retrieval" | 
               "tool_executor" | "sub_agent" | "response_generator";
    node_name: string;
    input: any;
}

interface NodeEndEvent {
    node_id: string;
    output: any;
    duration_ms: number;
    tokens_used: number;
}

// 执行明细数据结构
interface ExecutionDetail {
    step: number;
    node_type: string;
    node_name: string;
    input: any;
    output: any;
    duration_ms: number;
    tokens_used: number;

### A. 技术栈总览

| 层级 | 技术 | 版本 | 用途 |
|------|------|------|------|
| 前端框架 | Vue 3 | 3.4+ | Composition API |
| UI组件库 | Element Plus | 2.4+ | 企业级组件 |
| 构建工具 | Vite | 5.x | 极速构建 |
| 状态管理 | Pinia | 2.x | 状态管理 |
| 后端框架 | FastAPI | 0.115+ | 异步API |
| Agent引擎 | LangGraph | 0.2+ | 工作流编排 |
| LLM框架 | LangChain | 0.3+ | LLM集成 |
| 数据库 | PostgreSQL | 16+ | 持久化 |
| 缓存 | Redis | 7+ | 缓存/状态 |
| 向量库 | Milvus | 2.x | 向量检索 |
| 容器化 | Docker | - | 部署 |
| 编排 | Kubernetes | - | 生产部署 |

### B. 核心设计决策

| 决策 | 选择 | 原因 |
|------|------|------|
| 前端框架 | Vue3 (非React) | 更接近原系统，Element Plus生态 |
| 微服务划分 | 5个服务 | 职责分离，独立扩展 |
| Agent引擎 | LangGraph | 可视化工作流，过程可追溯 |
| 向量库 | Milvus | 高性能向量检索 |
| 配置存储 | JSONB | 灵活的Agent配置存储 |
